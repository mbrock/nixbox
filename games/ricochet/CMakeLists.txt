cmake_minimum_required(VERSION 3.21)
project(ricochet LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(SDL3 CONFIG REQUIRED)
find_package(box2d CONFIG REQUIRED)
find_package(imgui CONFIG REQUIRED)

add_library(ricochet_sim src/simulation.cpp)
target_include_directories(ricochet_sim PUBLIC src)
target_link_libraries(ricochet_sim PUBLIC box2d::box2d)
target_compile_definitions(ricochet_sim PRIVATE NOMINMAX)
add_executable(ricochet src/main.cpp)
target_link_libraries(ricochet PRIVATE ricochet_sim SDL3::SDL3 imgui::imgui)
target_compile_definitions(ricochet PRIVATE NOMINMAX)
if(WIN32)
  set_target_properties(ricochet PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
install(TARGETS ricochet RUNTIME DESTINATION bin)

include(CTest)
if(BUILD_TESTING)
  add_executable(simulation-test tests/simulation_test.cpp)
  target_link_libraries(simulation-test PRIVATE ricochet_sim)
  add_test(NAME simulation COMMAND simulation-test)
endif()
