cmake_minimum_required(VERSION 3.20)
project(ghostty_vt_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(SDL3 CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(VT REQUIRED IMPORTED_TARGET libghostty-vt)
set_source_files_properties(consumer.c PROPERTIES COMPILE_DEFINITIONS "main=ghostty_vt_probe_main")
add_executable(ghostty-vt-probe WIN32 main.cpp consumer.c)
target_link_libraries(ghostty-vt-probe PRIVATE SDL3::SDL3 PkgConfig::VT)
install(TARGETS ghostty-vt-probe RUNTIME DESTINATION bin)
