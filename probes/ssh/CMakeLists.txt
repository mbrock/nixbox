cmake_minimum_required(VERSION 3.20)
project(ssh_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(SDL3 CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SSH REQUIRED IMPORTED_TARGET libssh2)
add_executable(ssh-probe WIN32 main.cpp)
target_compile_definitions(ssh-probe PRIVATE NOMINMAX)
target_link_libraries(ssh-probe PRIVATE SDL3::SDL3 PkgConfig::SSH)
install(TARGETS ssh-probe RUNTIME DESTINATION bin)
