cmake_minimum_required(VERSION 3.21)
project(nxt-network LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(NXT_WEBSOCKET "Host the portable WebSocket probe instead of the network probe" OFF)
if(NXT_WEBSOCKET)
  set(probe_name nxt-websocket)
else()
  set(probe_name nxt-network)
endif()

find_package(SDL3 CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(NXT REQUIRED IMPORTED_TARGET nxtrt-iocp)

# Run the upstream probe unchanged, inside an Xbox-launchable SDL host.
configure_file("${NXT_PROBE_SOURCE}" portable-probe.cpp COPYONLY)
add_executable(${probe_name} WIN32 main.cpp)
target_compile_definitions(${probe_name} PRIVATE NXT_WEBSOCKET=$<BOOL:${NXT_WEBSOCKET}>)
target_include_directories(${probe_name} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(${probe_name} PRIVATE SDL3::SDL3 PkgConfig::NXT)
install(TARGETS ${probe_name} RUNTIME DESTINATION bin)
